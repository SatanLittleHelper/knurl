# Задача 4: Авторизация — регистрация и вход

**Что строим:** `POST /auth/register` и `POST /auth/login` — возвращают JWT-токен.

---

## Как работает авторизация

1. Пользователь регистрируется → сервер хэширует пароль и сохраняет в БД → возвращает JWT
2. Пользователь входит → сервер ищет пользователя, проверяет пароль → возвращает JWT
3. Дальше клиент передаёт JWT в заголовке `Authorization: Bearer <token>` при каждом запросе

**Зачем хэшировать пароль?**
Если БД утечёт — злоумышленник не получит пароли пользователей. bcrypt — медленный алгоритм хэширования специально для паролей: даже если хэш утёк, перебор займёт годы.

**Что такое JWT (JSON Web Token)?**
Токен из трёх частей: заголовок, данные (payload), подпись. Данные (например, `user_id`) хранятся прямо в токене — серверу не нужно ходить в БД чтобы понять кто делает запрос. Подпись гарантирует что токен не подделан.

---

## Как устроен код

В Go принято разделять слои:
- **handler** — принимает HTTP-запрос, валидирует входные данные, вызывает service, возвращает HTTP-ответ
- **service** — бизнес-логика: хэширование, запросы в БД, выдача токенов

Handler не знает как работает service, service не знает что его вызывает HTTP — это делает код легко тестируемым.

---

## GORM-модели

GORM работает с Go-структурами. Поля структуры соответствуют колонкам таблицы. Теги (`gorm:"..."`) настраивают это соответствие.

```go
type User struct {
    ID           uuid.UUID `gorm:"type:uuid;default:gen_random_uuid();primaryKey"`
    Email        string    `gorm:"uniqueIndex;not null"`
    PasswordHash string    `gorm:"column:password_hash;not null"`
    CreatedAt    time.Time
    UpdatedAt    time.Time
}
```

GORM автоматически обрабатывает `CreatedAt` и `UpdatedAt` — устанавливает при создании и обновлении.

import (
	"time"

	"github.com/google/uuid"

// User — GORM-модель пользователя.
// Теги gorm:"..." говорят GORM как работать с этим полем:
// - type:uuid — тип колонки в PostgreSQL
// - default:gen_random_uuid() — PostgreSQL генерирует UUID автоматически
// - primaryKey — это первичный ключ
// - uniqueIndex — создать уникальный индекс
// - not null — запретить NULL
// - column:password_hash — имя колонки в БД (иначе GORM использует snake_case от имени поля)
type User struct {
	ID           uuid.UUID `gorm:"type:uuid;default:gen_random_uuid();primaryKey"`
	Email        string    `gorm:"uniqueIndex;not null"`
	PasswordHash string    `gorm:"column:password_hash;not null"`
	CreatedAt    time.Time
	UpdatedAt    time.Time
}

- [ ] **Шаг 2: Создать `internal/auth/service.go`**

```go
package auth

import (
	"context"
	"errors"
	"time"

	"github.com/golang-jwt/jwt/v5"
	"github.com/google/uuid"
	"gorm.io/gorm"
	"golang.org/x/crypto/bcrypt"
)

// Sentinel errors — ошибки с фиксированным смыслом.
// Handler проверяет errors.Is(err, ErrEmailTaken) чтобы вернуть нужный HTTP-код.
var ErrInvalidCredentials = errors.New("invalid credentials")
var ErrEmailTaken = errors.New("email already taken")

// Service содержит бизнес-логику авторизации.
// Получает *gorm.DB из main через конструктор — это и есть dependency injection.
type Service struct {
	db        *gorm.DB
	jwtSecret []byte
}

func NewService(db *gorm.DB, jwtSecret string) *Service {
	return &Service{db: db, jwtSecret: []byte(jwtSecret)}
}

// Register создаёт нового пользователя и возвращает JWT.
func (s *Service) Register(_ context.Context, email, password string) (string, error) {
	// bcrypt.GenerateFromPassword — хэширует пароль.
	// bcrypt.DefaultCost = 10 — количество раундов хэширования.
	// Чем выше cost — тем медленнее брутфорс, но и регистрация медленнее.
	hash, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
	if err != nil {
		return "", err
	}

	user := User{Email: email, PasswordHash: string(hash)}

	// db.Create(&user) — INSERT INTO users (...) VALUES (...)
	// GORM заполняет user.ID и user.CreatedAt из ответа БД.
	result := s.db.Create(&user)
	if result.Error != nil {
		// Проверяем на нарушение уникальности (email уже занят)
		if isUniqueViolation(result.Error) {
			return "", ErrEmailTaken
		}
		return "", result.Error
	}

	return s.issueToken(user.ID)
}

// Login проверяет учётные данные и возвращает JWT.
func (s *Service) Login(_ context.Context, email, password string) (string, error) {
	var user User

	// db.Where("email = ?", email).First(&user) — SELECT * FROM users WHERE email = ? LIMIT 1
	// GORM защищает от SQL-инъекций через параметризованные запросы (? заменяется безопасно)
	result := s.db.Where("email = ?", email).First(&user)
	if result.Error != nil {
		// Не раскрываем клиенту что пользователь не найден — возвращаем общую ошибку
		return "", ErrInvalidCredentials
	}

	// bcrypt.CompareHashAndPassword — сравнивает пароль с хэшем.
	// Нельзя просто сравнить строки — нужно именно эта функция.
	if err := bcrypt.CompareHashAndPassword([]byte(user.PasswordHash), []byte(password)); err != nil {
		return "", ErrInvalidCredentials
	}

	return s.issueToken(user.ID)
}

// issueToken создаёт подписанный JWT с user_id внутри.
func (s *Service) issueToken(userID uuid.UUID) (string, error) {
	// MapClaims — данные внутри токена (payload).
	// "sub" (subject) — стандартное поле JWT для идентификатора пользователя.
	// "exp" (expiration) — когда токен истекает. Здесь 30 дней.
	claims := jwt.MapClaims{
		"sub": userID.String(),
		"exp": time.Now().Add(30 * 24 * time.Hour).Unix(),
	}

	// Подписываем токен нашим секретным ключом (HMAC-SHA256).
	// Без ключа никто не может создать валидный токен.
	return jwt.NewWithClaims(jwt.SigningMethodHS256, claims).SignedString(s.jwtSecret)
}

// isUniqueViolation проверяет является ли ошибка нарушением уникального индекса.
func isUniqueViolation(err error) bool {
	return err != nil && (contains(err.Error(), "unique") || contains(err.Error(), "23505"))
}

func contains(s, sub string) bool {
	for i := 0; i <= len(s)-len(sub); i++ {
		if s[i:i+len(sub)] == sub {
			return true
		}
	}
	return false
}
```

- [ ] **Шаг 3: Создать `internal/auth/handler.go`**

```go
package auth

import (
	"encoding/json"
	"errors"
	"net/http"
)

// Handler отвечает за HTTP-слой авторизации.
// Валидирует запросы, вызывает Service, формирует HTTP-ответы.
type Handler struct {
	svc *Service
}

func NewHandler(svc *Service) *Handler {
	return &Handler{svc: svc}
}

// authRequest — структура тела запроса для регистрации и входа.
// Теги json:"..." говорят как называются поля в JSON.
type authRequest struct {
	Email    string `json:"email"`
	Password string `json:"password"`
}

// Register обрабатывает POST /auth/register
func (h *Handler) Register(w http.ResponseWriter, r *http.Request) {
	var req authRequest
	// json.NewDecoder(r.Body).Decode(&req) — читает JSON из тела запроса
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "неверный формат запроса", http.StatusBadRequest)
		return
	}
	if req.Email == "" || req.Password == "" {
		http.Error(w, "email и пароль обязательны", http.StatusBadRequest)
		return
	}

	token, err := h.svc.Register(r.Context(), req.Email, req.Password)
	if errors.Is(err, ErrEmailTaken) {
		http.Error(w, "email уже занят", http.StatusConflict) // 409 Conflict
		return
	}
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]string{"token": token})
}

// Login обрабатывает POST /auth/login
func (h *Handler) Login(w http.ResponseWriter, r *http.Request) {
	var req authRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "неверный формат запроса", http.StatusBadRequest)
		return
	}

	token, err := h.svc.Login(r.Context(), req.Email, req.Password)
	if errors.Is(err, ErrInvalidCredentials) {
		http.Error(w, "неверные учётные данные", http.StatusUnauthorized) // 401 Unauthorized
		return
	}
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]string{"token": token})
}
```

- [ ] **Шаг 4: Обновить `cmd/api/main.go` — подключить авторизацию**

```go
package main

import (
	"fmt"
	"log"
	"net/http"

	"github.com/go-chi/chi/v5"
	"github.com/joho/godotenv"
	"github.com/satanlittlehelper/knurl/backend/internal/auth"
	"github.com/satanlittlehelper/knurl/backend/internal/config"
	"github.com/satanlittlehelper/knurl/backend/internal/db"
)

func main() {
	_ = godotenv.Load()
	cfg := config.Load()

	database, err := db.Connect(cfg.DatabaseURL)
	if err != nil {
		log.Fatalf("db connect: %v", err)
	}

	// Создаём сервис авторизации, передавая БД и JWT-секрет.
	// Сервис сам не знает откуда взялась БД — просто получает готовый объект.
	authSvc := auth.NewService(database, cfg.JWTSecret)
	authHandler := auth.NewHandler(authSvc)

	r := chi.NewRouter()
	r.Get("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
	})

	// Публичные маршруты — без авторизации
	r.Post("/auth/register", authHandler.Register)
	r.Post("/auth/login", authHandler.Login)

	addr := fmt.Sprintf(":%s", cfg.Port)
	log.Printf("server started on %s", addr)
	log.Fatal(http.ListenAndServe(addr, r))
}
```
