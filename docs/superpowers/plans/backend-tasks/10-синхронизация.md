# Задача 10: Эндпоинт синхронизации

**Что строим:** `POST /sync` — принимает изменения с мобильного приложения, применяет их, возвращает изменения с сервера.

---

## Как работает офлайн-синхронизация

Мобильное приложение работает с локальным SQLite. Когда появляется интернет — отправляет накопившиеся изменения на сервер и получает изменения сделанные с других устройств.

**Протокол:**
1. Клиент отправляет: `{ changes: {...}, last_sync_at: "2026-05-01T10:00:00Z" }`
   - `changes` — всё что изменилось на клиенте с прошлой синхронизации
   - `last_sync_at` — когда была последняя синхронизация (сервер вернёт всё новее этой даты)
2. Сервер применяет изменения клиента
3. Сервер возвращает: `{ changes: {...}, synced_at: "..." }`
   - `changes` — всё что изменилось на сервере с `last_sync_at`
   - `synced_at` — клиент сохранит это как новый `last_sync_at`

**Разрешение конфликтов: last-write-wins**
Если одна и та же запись изменена и на клиенте, и на сервере — побеждает та у которой `updated_at` новее. Это простейший алгоритм, достаточный для v1 где у пользователя обычно одно устройство.

**Мягкое удаление в синхронизации:**
Если запись удалена (имеет `deleted_at`), она всё равно попадает в `changes`. Так клиент узнаёт что нужно удалить запись у себя. Если бы мы удаляли физически — клиент никогда не узнал бы об удалении.

- [ ] **Шаг 1: Создать `internal/sync/service.go`**

```go
package sync

import (
	"time"

	"github.com/google/uuid"
	"github.com/satanlittlehelper/knurl/backend/internal/plans"
	"github.com/satanlittlehelper/knurl/backend/internal/sessions"
	"github.com/satanlittlehelper/knurl/backend/internal/sets"
	"gorm.io/gorm"
	"gorm.io/gorm/clause"
)

// Changes содержит изменённые записи всех типов.
// Используется и в запросе (что прислал клиент) и в ответе (что отдаёт сервер).
type Changes struct {
	Plans    []plans.WorkoutPlan       `json:"plans"`
	Sessions []sessions.WorkoutSession `json:"sessions"`
	Sets     []sets.SetLog             `json:"sets"`
}

type SyncRequest struct {
	Changes    Changes   `json:"changes"`
	LastSyncAt time.Time `json:"last_sync_at"` // клиент помнит когда была последняя синхронизация
}

type SyncResponse struct {
	Changes  Changes   `json:"changes"`
	SyncedAt time.Time `json:"synced_at"` // клиент сохранит это как новый last_sync_at
}

type Service struct{ db *gorm.DB }

func NewService(db *gorm.DB) *Service { return &Service{db: db} }

// Apply применяет изменения клиента и возвращает изменения сервера.
func (s *Service) Apply(userID uuid.UUID, req SyncRequest) (SyncResponse, error) {
	now := time.Now()

	// Применяем изменения планов.
	// clause.OnConflict — что делать при конфликте (запись уже существует):
	// Обновляем поля ТОЛЬКО если updated_at клиента новее чем в БД (last-write-wins).
	if len(req.Changes.Plans) > 0 {
		for i := range req.Changes.Plans {
			req.Changes.Plans[i].UserID = userID // убеждаемся что план принадлежит правильному пользователю
		}
		result := s.db.Clauses(clause.OnConflict{
			Columns: []clause.Column{{Name: "id"}},
			// DoUpdates: обновить name, updated_at, deleted_at — но только если updated_at в запросе новее
			DoUpdates: clause.Assignments(map[string]any{
				"name":       gorm.Expr("CASE WHEN workout_plans.updated_at < EXCLUDED.updated_at THEN EXCLUDED.name ELSE workout_plans.name END"),
				"updated_at": gorm.Expr("CASE WHEN workout_plans.updated_at < EXCLUDED.updated_at THEN EXCLUDED.updated_at ELSE workout_plans.updated_at END"),
				"deleted_at": gorm.Expr("CASE WHEN workout_plans.updated_at < EXCLUDED.updated_at THEN EXCLUDED.deleted_at ELSE workout_plans.deleted_at END"),
			}),
		}).Create(&req.Changes.Plans)
		if result.Error != nil {
			return SyncResponse{}, result.Error
		}
	}

	if len(req.Changes.Sessions) > 0 {
		for i := range req.Changes.Sessions {
			req.Changes.Sessions[i].UserID = userID
		}
		result := s.db.Clauses(clause.OnConflict{
			Columns: []clause.Column{{Name: "id"}},
			DoUpdates: clause.Assignments(map[string]any{
				"finished_at": gorm.Expr("CASE WHEN workout_sessions.updated_at < EXCLUDED.updated_at THEN EXCLUDED.finished_at ELSE workout_sessions.finished_at END"),
				"updated_at":  gorm.Expr("CASE WHEN workout_sessions.updated_at < EXCLUDED.updated_at THEN EXCLUDED.updated_at ELSE workout_sessions.updated_at END"),
				"deleted_at":  gorm.Expr("CASE WHEN workout_sessions.updated_at < EXCLUDED.updated_at THEN EXCLUDED.deleted_at ELSE workout_sessions.deleted_at END"),
			}),
		}).Create(&req.Changes.Sessions)
		if result.Error != nil {
			return SyncResponse{}, result.Error
		}
	}

	if len(req.Changes.Sets) > 0 {
		result := s.db.Clauses(clause.OnConflict{
			Columns: []clause.Column{{Name: "id"}},
			DoUpdates: clause.Assignments(map[string]any{
				"reps":       gorm.Expr("CASE WHEN set_logs.updated_at < EXCLUDED.updated_at THEN EXCLUDED.reps ELSE set_logs.reps END"),
				"weight":     gorm.Expr("CASE WHEN set_logs.updated_at < EXCLUDED.updated_at THEN EXCLUDED.weight ELSE set_logs.weight END"),
				"updated_at": gorm.Expr("CASE WHEN set_logs.updated_at < EXCLUDED.updated_at THEN EXCLUDED.updated_at ELSE set_logs.updated_at END"),
				"deleted_at": gorm.Expr("CASE WHEN set_logs.updated_at < EXCLUDED.updated_at THEN EXCLUDED.deleted_at ELSE set_logs.deleted_at END"),
			}),
		}).Create(&req.Changes.Sets)
		if result.Error != nil {
			return SyncResponse{}, result.Error
		}
	}

	// Получаем всё что изменилось на сервере с момента последней синхронизации клиента.
	// Unscoped() — включить мягко удалённые записи (клиент должен знать что удалено).
	var serverPlans []plans.WorkoutPlan
	s.db.Unscoped().Where("user_id = ? AND updated_at > ?", userID, req.LastSyncAt).Find(&serverPlans)

	var serverSessions []sessions.WorkoutSession
	s.db.Unscoped().Where("user_id = ? AND updated_at > ?", userID, req.LastSyncAt).Find(&serverSessions)

	// Подходы не имеют user_id напрямую — находим через сессии пользователя
	var serverSets []sets.SetLog
	s.db.Unscoped().
		Joins("JOIN workout_sessions ON workout_sessions.id = set_logs.session_id").
		Where("workout_sessions.user_id = ? AND set_logs.updated_at > ?", userID, req.LastSyncAt).
		Find(&serverSets)

	return SyncResponse{
		Changes: Changes{
			Plans:    serverPlans,
			Sessions: serverSessions,
			Sets:     serverSets,
		},
		SyncedAt: now,
	}, nil
}
```

- [ ] **Шаг 2: Создать `internal/sync/handler.go`**

```go
package sync

import (
	"encoding/json"
	"net/http"

	"github.com/satanlittlehelper/knurl/backend/internal/auth"
)

type Handler struct{ svc *Service }

func NewHandler(svc *Service) *Handler { return &Handler{svc: svc} }

// Sync обрабатывает POST /sync
func (h *Handler) Sync(w http.ResponseWriter, r *http.Request) {
	userID, _ := auth.UserIDFromCtx(r.Context())

	var req SyncRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "bad request", http.StatusBadRequest)
		return
	}

	resp, err := h.svc.Apply(userID, req)
	if err != nil {
		http.Error(w, "internal error", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(resp)
}
```
