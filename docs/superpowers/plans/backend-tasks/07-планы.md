# Задача 7: CRUD планов тренировок

**Что строим:** Маршруты для управления планами тренировок, днями и упражнениями в днях.

**Маршруты:**
- `GET /plans` — список планов пользователя
- `POST /plans` — создать план
- `DELETE /plans/:id` — удалить план (мягко)
- `GET /plans/:id/days` — дни плана
- `POST /plans/:id/days` — добавить день
- `POST /plans/:id/days/:dayId/exercises` — добавить упражнение в день

---

## Мягкое удаление (Soft Delete)

Мягкое удаление — запись не удаляется из БД, а получает `deleted_at = NOW()`. Зачем:
- Данные можно восстановить
- Синхронизация с мобильным приложением: телефон должен узнать что запись удалена (если удалить физически — нельзя передать "эта запись удалена")

GORM поддерживает мягкое удаление через `gorm.DeletedAt`. Если поле есть в модели — GORM автоматически:
- Добавляет `WHERE deleted_at IS NULL` ко всем запросам
- При `db.Delete(&model)` устанавливает `deleted_at = NOW()` вместо DELETE

```go
package plans

import (
	"time"

	"github.com/google/uuid"
	"gorm.io/gorm"
)

// WorkoutPlan — план тренировок пользователя (например "Сплит 3 дня").
// gorm.DeletedAt — специальный тип GORM для мягкого удаления.
// Когда он есть в модели — GORM автоматически фильтрует удалённые записи.
type WorkoutPlan struct {
	ID        uuid.UUID      `gorm:"type:uuid;default:gen_random_uuid();primaryKey" json:"id"`
	UserID    uuid.UUID      `gorm:"type:uuid;not null"                             json:"user_id"`
	Name      string         `gorm:"not null"                                       json:"name"`
	CreatedAt time.Time      `                                                      json:"created_at"`
	UpdatedAt time.Time      `                                                      json:"updated_at"`
	DeletedAt gorm.DeletedAt `gorm:"index"                                          json:"deleted_at,omitempty"`
}

// WorkoutDay — один день плана (например "День А — грудь и трицепс").
// day_of_week: 0=воскресенье, 1=понедельник, ..., 6=суббота.
type WorkoutDay struct {
	ID        uuid.UUID      `gorm:"type:uuid;default:gen_random_uuid();primaryKey" json:"id"`
	PlanID    uuid.UUID      `gorm:"type:uuid;not null"                             json:"plan_id"`
	DayOfWeek int            `gorm:"not null"                                       json:"day_of_week"`
	UpdatedAt time.Time      `                                                      json:"updated_at"`
	DeletedAt gorm.DeletedAt `gorm:"index"                                          json:"deleted_at,omitempty"`
}

// PlannedExercise — упражнение в дне плана с целевыми параметрами.
type PlannedExercise struct {
	ID           uuid.UUID      `gorm:"type:uuid;default:gen_random_uuid();primaryKey" json:"id"`
	DayID        uuid.UUID      `gorm:"type:uuid;not null"                             json:"day_id"`
	ExerciseID   string         `gorm:"not null"                                       json:"exercise_id"`
	Sets         int            `gorm:"not null"                                       json:"sets"`
	Reps         int            `gorm:"not null"                                       json:"reps"`
	TargetWeight *float64       `gorm:"column:target_weight"                           json:"target_weight,omitempty"` // указатель — может быть nil
	Position     int            `gorm:"not null"                                       json:"position"`
	UpdatedAt    time.Time      `                                                      json:"updated_at"`
	DeletedAt    gorm.DeletedAt `gorm:"index"                                          json:"deleted_at,omitempty"`
}
```

- [ ] **Шаг 2: Создать `internal/plans/service.go`**

```go
package plans

import (
	"github.com/google/uuid"
	"gorm.io/gorm"
)

type Service struct {
	db *gorm.DB
}

func NewService(db *gorm.DB) *Service { return &Service{db: db} }

// ListPlans возвращает планы пользователя (без удалённых — GORM фильтрует автоматически).
func (s *Service) ListPlans(userID uuid.UUID) ([]WorkoutPlan, error) {
	var plans []WorkoutPlan
	// db.Where("user_id = ?", userID).Order("created_at DESC").Find(&plans)
	// → SELECT * FROM workout_plans WHERE user_id = ? AND deleted_at IS NULL ORDER BY created_at DESC
	result := s.db.Where("user_id = ?", userID).Order("created_at DESC").Find(&plans)
	return plans, result.Error
}

// CreatePlan создаёт новый план.
func (s *Service) CreatePlan(userID uuid.UUID, name string) (WorkoutPlan, error) {
	plan := WorkoutPlan{UserID: userID, Name: name}
	// db.Create(&plan) — INSERT INTO workout_plans (...)
	// GORM заполняет plan.ID из ответа БД
	result := s.db.Create(&plan)
	return plan, result.Error
}

// DeletePlan мягко удаляет план (устанавливает deleted_at).
// Проверяем user_id чтобы пользователь не мог удалить чужой план.
func (s *Service) DeletePlan(id, userID uuid.UUID) error {
	// db.Where(...).Delete(&WorkoutPlan{}) — UPDATE workout_plans SET deleted_at = NOW() WHERE ...
	result := s.db.Where("id = ? AND user_id = ?", id, userID).Delete(&WorkoutPlan{})
	return result.Error
}

// ListDays возвращает дни плана.
func (s *Service) ListDays(planID uuid.UUID) ([]WorkoutDay, error) {
	var days []WorkoutDay
	result := s.db.Where("plan_id = ?", planID).Order("day_of_week").Find(&days)
	return days, result.Error
}

// CreateDay добавляет день в план.
func (s *Service) CreateDay(planID uuid.UUID, dayOfWeek int) (WorkoutDay, error) {
	day := WorkoutDay{PlanID: planID, DayOfWeek: dayOfWeek}
	result := s.db.Create(&day)
	return day, result.Error
}

// AddExercise добавляет упражнение в день.
func (s *Service) AddExercise(ex PlannedExercise) (PlannedExercise, error) {
	result := s.db.Create(&ex)
	return ex, result.Error
}
```

- [ ] **Шаг 3: Создать `internal/plans/handler.go`**

```go
package plans

import (
	"encoding/json"
	"net/http"

	"github.com/go-chi/chi/v5"
	"github.com/google/uuid"
	"github.com/satanlittlehelper/knurl/backend/internal/auth"
)

type Handler struct{ svc *Service }

func NewHandler(svc *Service) *Handler { return &Handler{svc: svc} }

func (h *Handler) ListPlans(w http.ResponseWriter, r *http.Request) {
	userID, _ := auth.UserIDFromCtx(r.Context())
	plans, err := h.svc.ListPlans(userID)
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}
	respond(w, plans)
}

func (h *Handler) CreatePlan(w http.ResponseWriter, r *http.Request) {
	userID, _ := auth.UserIDFromCtx(r.Context())
	var body struct {
		Name string `json:"name"`
	}
	if err := json.NewDecoder(r.Body).Decode(&body); err != nil || body.Name == "" {
		http.Error(w, "name обязательное поле", http.StatusBadRequest)
		return
	}
	plan, err := h.svc.CreatePlan(userID, body.Name)
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusCreated) // 201 Created
	respond(w, plan)
}

func (h *Handler) DeletePlan(w http.ResponseWriter, r *http.Request) {
	userID, _ := auth.UserIDFromCtx(r.Context())
	// chi.URLParam(r, "id") — читает параметр из URL, например /plans/abc123 → "abc123"
	id, err := uuid.Parse(chi.URLParam(r, "id"))
	if err != nil {
		http.Error(w, "некорректный id", http.StatusBadRequest)
		return
	}
	if err := h.svc.DeletePlan(id, userID); err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusNoContent) // 204 No Content — успех без тела ответа
}

func (h *Handler) ListDays(w http.ResponseWriter, r *http.Request) {
	planID, err := uuid.Parse(chi.URLParam(r, "id"))
	if err != nil {
		http.Error(w, "некорректный id", http.StatusBadRequest)
		return
	}
	days, err := h.svc.ListDays(planID)
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}
	respond(w, days)
}

func (h *Handler) CreateDay(w http.ResponseWriter, r *http.Request) {
	planID, err := uuid.Parse(chi.URLParam(r, "id"))
	if err != nil {
		http.Error(w, "некорректный id", http.StatusBadRequest)
		return
	}
	var body struct {
		DayOfWeek int `json:"day_of_week"`
	}
	if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
		http.Error(w, "неверный формат запроса", http.StatusBadRequest)
		return
	}
	day, err := h.svc.CreateDay(planID, body.DayOfWeek)
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusCreated)
	respond(w, day)
}

func (h *Handler) AddExercise(w http.ResponseWriter, r *http.Request) {
	dayID, err := uuid.Parse(chi.URLParam(r, "dayId"))
	if err != nil {
		http.Error(w, "некорректный day id", http.StatusBadRequest)
		return
	}
	var body PlannedExercise
	if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
		http.Error(w, "неверный формат запроса", http.StatusBadRequest)
		return
	}
	body.DayID = dayID
	ex, err := h.svc.AddExercise(body)
	if err != nil {
		http.Error(w, "внутренняя ошибка", http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusCreated)
	respond(w, ex)
}

// respond — вспомогательная функция: устанавливает Content-Type и кодирует ответ в JSON.
func respond(w http.ResponseWriter, v any) {
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(v)
}
```
